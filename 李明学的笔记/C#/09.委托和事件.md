# <span style="color: #FFA000;">委托和事件</span> 
### 委托
```c#
// it is similar with function point
public delegate void DelegateFunc();
DelegateFunc delegate_func = new DelegateFunc(func);
delegate_func += func2;
delegate_func -= func2;
delegate_func();
void func1() {
    ;
}
void func2() {
    ;
}
void func3(DelegateFunc func) {
    func();
}
```
### 匿名函数
```c#
//Lambda expression
DelegateFunc delegate_func = new DelegateFunc(() => []);
delegate_func(); 
DelegateFunc delegate_func = delegate (int a, int b) {
    return a + b;
}
```
### 泛型委托
```c#
//You don't need to create a delegate 
//Action does not have return type
//Usually using on thread
Action<int> action = new Action<int>(func_action);
//Func has return type, last type is return type
//Usually using on extention func
Func<int, string> func = new Func<int, string>(func_func);
void func_action(int i) {
    ;
}
string func_func(int i) {
    ;
}
//通用行为进行包装
int add(int a, int b) {
    return a + b;
}
int Operation(Func<int, int> func, sum[], from, to) }{
    rt = sum[from];
    for (i = from; i <= to; i++) {
        rt = func(rt, sum[i]);
    }
    return rt;
}
//Call
int result = Operation(add, sum[], from, to);
```
### 事件
```c#
//可以理解为封装的委托
public delegate void DelegateFunc();
public DelegateFunc _delegate;
public event DelegateFunc _event;
// event only support += and -=
_event += func;
_event -= func;
void func() {
    ;
}
```
