# <span style="color: #FFA000;">线程和Task</span> 
### 线程
```c#
using System.Threading;
Thread thread = new Thread(
    () => {
        // Only main thread can assign control directly.
        if (this.control.InvokeRequired) {
            this.control.invoke(
                new Action<string>(data => {this.control.Text = data}),
                "Hello World");
        }
    }
);
thread.IsBackground = true;
thread.Start();
thread.Join();//线程阻塞
```

### Task
```c#
//Task = Thread + ThreadPool
//It is similat with ThreadPool
Task task_factory = Task.Factory.StartNew(()=>{
    Thread.Sleep(5000);
});
//Split time-consuming work to here
task_factory.ContinueWith(t=>{
    this.control.Text = "";
}, TaskScheduler.FromCurrentSynchorizationConText());

Task task_start = new Task(()=>{
    Thread.Sleep(5000);
    this.control.Text = "";
});
// It's a time-consuming work
task.Start(TaskScheduler.FromCurrentSynchorizationConText());
task.Wait();//Task的阻塞，与Thread.Join一致
Task task_run = Task.Run(()=>{
    ;
});
Task.WaitAll(task_start, task_run); //Wait all target
Task.WaitAny(task_start, task_run); //Wait any of one
//The main thread does not need to wait
Task.WhenAll(task_start, task_run).ContinueWith(task_when_all=>{
    ;
});
Task.WhenAny(task_start, task_run).ContinueWith(task_when_any=>{
    ;
});

//It's the same as WaitAll
Task parrent_task = new Task(()=>{
    Task task_child_1 = new Task(()=>{
        ;
    }, TaskCreationOptions.AttachedToParrent);
    Task task_child_2 = new Task(()=>{
        ;
    }, TaskCreationOptions.AttachedToParrent);
    task_child_1.Start();
    task_child_2.Start();
});
parrent_task.Start();
try {
    parrent_task.Wait();
} catch(aggregateException ex) {
    ;
    ex.Handel(p=>{
        return False; //throw exception to up
    });
}
Task task_long_running = new Task(()=>{
    ;
}, TaskCreationOptions.LongRunning);
//easily cancel task, thread have to throw exception for cancel
CancellationTokenSource cts = new CancellationTokenSource();
Task task_factory = Task.Factory.StartNew(()=>{
    while(cts.IsCancellationRequested) {
        ;
    }
}, cts.Token);
cts.Token.Register(()=>{
    ;//Call it before cancel, can release some stuff before cancel
});
cts.Cancel();
// timeout
cts.CancelAfter(3000);
```

### async & await
