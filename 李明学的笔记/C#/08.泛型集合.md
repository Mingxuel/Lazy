### 泛型(Generic)
```c#
public static void Swap<T>(ref T a, ref T b) {
    T temp = a;
    a = b;
    b = temp;
}
class MyGeneric<T> {
    private T data;
    public MyGeneric() {
        // when you are not make sure if it is a value type or ref type
        data = default(T); 
    }
    public T Add(T a, T b) {
        // errors may occur during compilation
        // use dynamic to skip build error
        runtime variable, build is ok.
        dynamic d_a = a;
        dynamic d_b = b;
        return d_a + d_b;
    }
}
```
