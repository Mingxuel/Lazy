### 运算符
``` c#
int a = 10;
a = a + 10;                 // symbol '=', is assigned to
a++;                        // post-increment
++a;                        // pre-increment
a += 1;                     // increment by 1 | add 1 to
a--;                        // post-decrement
--a;                        // pre-decrement
a -= 1;                     //  decrement by 1 | subtract 1 from

a >	b;                      // greater than
a <	b;                      // less than
a >= b;                     // greater than or equal to
a <= b;                     // less than or equal to
a == b;                     // equals / is equal to
a != b;                     // not equal to

a = a > b ? a : b;          // ternary operator | conditional operator

a && b;                     // logical AND | conditional AND
a || b;                     // logical OR | conditional OR
a ! b;                      // logical NOT
a ^ b;                      // logical XOR | exclusive OR

a & b;                      // bitwise AND
a | b;                      // bitwise OR
a ^ b;                      // bitwise XOR
~a;                         // bitwise NOT
a << b;                     // bitwise left shift
a >> b;                     // bitwise right shift
```

### 转义字符
```c#
\                           // Escape Character
'\*'                        // Escape Sequence
\n	                        // Newline	        换行	    光标移至下一行
\t	                        // Tab	            制表符	    插入水平制表符
\\	                        // Backslash	    反斜杠本身  表示一个反斜杠 \
\"	                        // Double Quote	    双引号	    在字符串中插入双引号
\'	                        // Single Quote	    单引号	    在字符串中插入单引号
\r	                        // Carriage Return	回车	    光标移至行首
\b	                        // Backspace	    退格	    删除前一个字符
\0	                        // Null Character	空字符	    表示 ASCII 码 0
```
